Tạo đối tượng đơn hàng const orderData = { name, phone, product, time, type, distance, fee }; // 2. Gửi thông báo về Telegram cho chủ shop sendTelegramNotification(orderData); // 3. Phát âm thanh thông báo trên máy tính (nếu đang mở web) playNotificationSound(); // 4. Thông báo cho khách alert("Cảm ơn " + name + "! Đơn hàng đã được gửi tới hệ thống của Thu Mến."); // Cập nhật trạng thái hiển thị adminUpdateStatus("Khách vừa đặt đơn mới"); } // --- GIỮ NGUYÊN CÁC HÀM CŨ (renderProducts, adminAddProduct, toggleShipping...) --- function renderProducts() { const customerSelect = document.getElementById('customerProductSelect'); const adminList = document.getElementById('adminProductList'); customerSelect.innerHTML = ''; adminList.innerHTML = ''; products.forEach((p, index) => { let option = document.createElement('option'); option.value = p.name; option.text = `${p.name} - ${p.price}`; customerSelect.appendChild(option); adminList.innerHTML += `
${p.name} (${p.price})
`; }); localStorage.setItem('thuMenProducts', JSON.stringify(products)); } function adminAddProduct() { const name = document.getElementById('newProdName').value; const price = document.getElementById('newProdPrice').value; if(name && price) { products.push({name, price}); renderProducts(); } } function adminDeleteProduct(index) { products.splice(index, 1); renderProducts(); } function toggleShipping() { const type = document.getElementById('deliveryType').value; document.getElementById('shipDetails').style.display = (type === 'ship') ?
